2 - Diferença entre let e const

As duas declaram variáveis com escopo de bloco (valem só dentro das chaves onde foram criadas). A diferença é que uma variável declarada com let pode receber um novo valor depois, enquanto uma declarada com const não pode ser reatribuída e precisa ser inicializada na declaração. (Se a const guarda um objeto ou array, o conteúdo interno ainda pode mudar; o que não muda é a referência.)

// let: valor que muda, como um contador ou um acumulador
let soma = 0;
for (let i = 1; i <= 5; i++) {
    soma += i;
}

// const: valor que não deve mudar, como a média para aprovação
const MEDIA_APROVACAO = 6;
const botao = document.getElementById("calcular");

Regra prática: usar const por padrão e let apenas quando a variável realmente precisar ser reatribuída.

3 - Incluindo um arquivo JavaScript externo

Usando a tag <script> com o atributo src apontando para o arquivo:

<script src="script.js"></script>

O local mais comum é no final do <body>, logo antes de </body>. Assim o navegador carrega e mostra todo o HTML primeiro, a página aparece mais rápido para o usuário, e quando o script roda todos os elementos já existem, então funções como document.getElementById encontram o que precisam. Outra opção moderna é colocar no <head> com o atributo defer, que tem o mesmo efeito de esperar o HTML ser lido.

4 - Leia dois números e apresente a multiplicação deles.

Arquivo: js/exercicio1.js

// 1 - Leia dois números e apresente a multiplicação deles.
function exercicio1() {
    const numero1 = Number(prompt("Digite o primeiro número:"));
    const numero2 = Number(prompt("Digite o segundo número:"));

    if (isNaN(numero1) || isNaN(numero2)) {
        alert("Digite apenas números válidos.");
        return;
    }

    alert(numero1 + " x " + numero2 + " = " + (numero1 * numero2));
}

5 - Leia um número e informe se é par ou ímpar e se é positivo ou negativo.

Arquivo: js/exercicio2.js

// 2 - Leia um número e informe se é par ou ímpar e se é positivo ou negativo.
function exercicio2() {
    const numero = Number(prompt("Digite um número inteiro:"));

    if (isNaN(numero) || !Number.isInteger(numero)) {
        alert("Digite um número inteiro válido.");
        return;
    }

    const paridade = numero % 2 === 0 ? "par" : "ímpar";
    let sinal;
    if (numero > 0) {
        sinal = "positivo";
    } else if (numero < 0) {
        sinal = "negativo";
    } else {
        sinal = "zero (nem positivo nem negativo)";
    }

    alert("O número " + numero + " é " + paridade + " e " + sinal + ".");
}

7 - Receba um número e mostre a tabuada.

Arquivo: js/exercicio4.js

// 4 - Receba um número e mostre a tabuada dele.
function exercicio4() {
    const numero = Number(prompt("Digite um número para ver a tabuada:"));

    if (isNaN(numero)) {
        alert("Digite um número válido.");
        return;
    }

    let tabuada = "Tabuada do " + numero + ":\n";
    for (let i = 1; i <= 10; i++) {
        tabuada += numero + " x " + i + " = " + (numero * i) + "\n";
    }

    alert(tabuada);
}

8 - Leia três notas, mostre a média e se o aluno está aprovado (média 6).

Arquivo: js/exercicio5.js

// 5 - Leia três notas, mostre a média e se o aluno foi aprovado (média 6).
function exercicio5() {
    const nota1 = Number(prompt("Digite a primeira nota:"));
    const nota2 = Number(prompt("Digite a segunda nota:"));
    const nota3 = Number(prompt("Digite a terceira nota:"));

    if (isNaN(nota1) || isNaN(nota2) || isNaN(nota3)) {
        alert("Digite notas válidas.");
        return;
    }

    const media = (nota1 + nota2 + nota3) / 3;
    const situacao = media >= 6 ? "Aprovado" : "Reprovado";

    alert("Média: " + media.toFixed(2) + "\nSituação: " + situacao);
}

9 - Leia um número de 1 a 7 e mostre o dia da semana.

Arquivo: js/exercicio6.js

// 6 - Leia um número de 1 a 7 e mostre o dia da semana (1 = domingo, 7 = sábado).
function exercicio6() {
    const numero = Number(prompt("Digite um número de 1 a 7:"));

    switch (numero) {
        case 1: alert("Domingo"); break;
        case 2: alert("Segunda-feira"); break;
        case 3: alert("Terça-feira"); break;
        case 4: alert("Quarta-feira"); break;
        case 5: alert("Quinta-feira"); break;
        case 6: alert("Sexta-feira"); break;
        case 7: alert("Sábado"); break;
        default: alert("Número inválido! Digite um valor de 1 a 7.");
    }
}

10 - Leia números até um negativo e mostre a média.

Arquivo: js/exercicio7.js

// 7 - Leia números até que um negativo seja digitado e mostre a média dos números digitados.
function exercicio7() {
    let soma = 0;
    let quantidade = 0;

    while (true) {
        const entrada = prompt("Digite um número (um número negativo encerra):");
        if (entrada === null) {
            break;
        }

        const numero = Number(entrada);
        if (isNaN(numero) || entrada.trim() === "") {
            alert("Valor inválido, tente novamente.");
            continue;
        }
        if (numero < 0) {
            break;
        }

        soma += numero;
        quantidade++;
    }

    if (quantidade === 0) {
        alert("Nenhum número válido foi digitado.");
    } else {
        alert("Você digitou " + quantidade + " número(s). Média: " + (soma / quantidade).toFixed(2));
    }
}
